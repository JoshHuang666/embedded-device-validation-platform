cmake_minimum_required(VERSION 3.16)
project(EmbeddedDeviceController VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(device_core firmware/src/device.c)
target_include_directories(device_core PUBLIC firmware/include)
add_executable(device_sim firmware/src/main.c)
target_link_libraries(device_sim PRIVATE device_core)
enable_testing()
add_executable(test_device tests/test_device.c)
target_link_libraries(test_device PRIVATE device_core)
add_test(NAME DeviceStateTest COMMAND test_device)

target_link_libraries(device_core PUBLIC m)

add_executable(test_voltage tests/test_voltage.c)
target_link_libraries(test_voltage PRIVATE device_core)
add_test(NAME VoltageTest COMMAND test_voltage)

target_sources(device_core PRIVATE firmware/src/power_model.c)

add_executable(test_power_model tests/test_power_model.c)
target_link_libraries(test_power_model PRIVATE device_core)
foreach(case_number RANGE 1 14)
    add_test(NAME PowerModel_TC${case_number} COMMAND test_power_model ${case_number})
endforeach()
